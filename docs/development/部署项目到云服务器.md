## 打包部署

```bash
# 打开 IDEA 终端（或者在项目根目录打开 CMD）
# 执行打包命令（跳过测试）
mvn clean package "-Dmaven.test.skip=true"



# 在 PowerShell 中逐个复制
# 或者打开文件资源管理器，到各个 target 目录下，把这 9 个核心 jar 包复制到一个文件夹里就行！
# 先创建目录
mkdir -p E:\codebase\daijia-release

# 复制模块
cp E:\codebase\leshang-daijia-20260104\server-gateway\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\service\service-customer\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\service\service-driver\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\service\service-order\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\service\service-map\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\service\service-rules\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\service\service-dispatch\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\web\web-customer\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\web\web-driver\target\*.jar E:\codebase\daijia-release\
cp E:\codebase\leshang-daijia-20260104\xxl-job-master\xxl-job-admin\target\*.jar E:\codebase\daijia-release\

# 打包
cd E:\codebase
tar -czf daijia.tar.gz daijia-release

# 服务器上创建 /app 目录
mkdir -p /app
# 上传到服务器
scp daijia.tar.gz root@192.168.101.12:/app/

# 2. 强制删除整个目录（不询问）
#rm -rf daijia
#rm -rf daijia-release
#rm -f daijia.tar.gz

# 在服务器上解压
cd /app
tar -xzf daijia.tar.gz
# 1. 先创建 daijia 目录
mkdir -p daijia

# 2. 再移动文件
mv daijia-release/* daijia/

# 3. 删除空的 release 目录
rmdir daijia-release


# 在/app/daijia/目录下创建启动脚本：
vim start-all.sh

# 添加如下：{
#!/bin/bash
echo "Starting Daijia services..."

# 创建日志目录
mkdir -p /var/log/daijia

# 按依赖顺序启动（注意 jar 文件都在当前目录）
cd /app/daijia

# 先启动 xxl-job-admin
nohup java -jar xxl-job-admin-2.3.0.jar --spring.profiles.active=dev > /var/log/daijia/xxl-job-admin.log 2>&1 &
echo "xxl-job-admin started"
sleep 10

# 启动基础服务
nohup java -jar service-customer-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/customer.log 2>&1 &
echo "service-customer started"
sleep 5

nohup java -jar service-driver-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/driver.log 2>&1 &
echo "service-driver started"
sleep 5

nohup java -jar service-order-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/order.log 2>&1 &
echo "service-order started"
sleep 5

nohup java -jar service-rules-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/rules.log 2>&1 &
echo "service-rules started"
sleep 5

nohup java -jar service-map-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/map.log 2>&1 &
echo "service-map started"
sleep 5

nohup java -jar service-dispatch-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/dispatch.log 2>&1 &
echo "service-dispatch started"
sleep 5

# 启动 web 模块
nohup java -jar web-customer-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/web-customer.log 2>&1 &
echo "web-customer started"
sleep 5

nohup java -jar web-driver-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/web-driver.log 2>&1 &
echo "web-driver started"
sleep 5

# 最后启动网关
nohup java -jar server-gateway-1.0.0.jar --spring.profiles.active=dev > /var/log/daijia/gateway.log 2>&1 &
echo "server-gateway started"

echo "All services started!"
echo "Check logs in /var/log/daijia/"
# }

# 给执行权限
chmod +x start-all.sh

# 在/app/daijia/目录下创建停止脚本
vim stop-all.sh

# 添加如下：{
#!/bin/bash
echo "Stopping Daijia services..."

# 按顺序停止（先停网关，再停其他）
pkill -f server-gateway-1.0.0.jar
sleep 2

pkill -f web-customer-1.0.0.jar
pkill -f web-driver-1.0.0.jar
sleep 2

pkill -f service-customer-1.0.0.jar
pkill -f service-driver-1.0.0.jar
pkill -f service-order-1.0.0.jar
pkill -f service-map-1.0.0.jar
pkill -f service-rules-1.0.0.jar
pkill -f service-dispatch-1.0.0.jar
sleep 2

pkill -f xxl-job-admin-2.3.0.jar

# 检查是否还有残留
sleep 2
if pgrep -f "daijia|service-|web-|gateway" > /dev/null; then
    echo "Force killing remaining processes..."
    pkill -9 -f "daijia|service-|web-|gateway"
fi

echo "All services stopped"
# }

chmod +x /app/daijia/stop-all.sh
```
