service.interceptors.response.use(
  (response) => {
    var _a;
    
    // =========== 关键修复：检查并更新Token ===========
    // 1. 检查响应头中是否有新Token
    const newToken = response.header?.['new-token'] || response.header?.['New-Token'];
    if (newToken) {
      console.log('✅ 收到新Token，更新本地存储', newToken.substring(0, 50) + '...');
      
      // 2. 保存新Token到本地存储
      utils_storage.setToken(newToken);
      
      // 3. 如果有用户store，可以更新store中的token
      if (store_modules_user.useUserStore) {
        const userStore = store_modules_user.useUserStore();
        if (userStore && userStore.setToken) {
          userStore.setToken(newToken);
        }
      }
      
      // 4. 可选：显示提示
      setTimeout(() => {
        common_vendor.index.showToast({
          title: '登录状态已更新',
          icon: 'success',
          duration: 2000
        });
      }, 300);
    }
    // =========== 关键修复结束 ===========
    
    const { data } = response;
    
    if (http_type.ResultEnum.EXPIRE.includes(data.code)) {
      (_a = store_modules_user.useUserStore()) == null ? void 0 : _a.$reset();
      common_vendor.index.showModal({
        title: "提示",
        content: "登录过期，请重新登录",
        success: function(res) {
          if (res.confirm) {
            common_vendor.index.clearStorageSync();
            common_vendor.index.redirectTo({
              url: "/pages/login/login"
            });
          } else if (res.cancel) {
            console.log("用户不想登陆");
          }
        }
      });
      return Promise.reject(data);
    }
    
    if (data.code && data.code !== http_type.ResultEnum.SUCCESS) {
      common_vendor.index.showToast({
        title: data.message || http_type.ResultEnum.ERRMESSAGE,
        icon: "error"
      });
      return Promise.reject(data);
    }
    return data;
  },
  (response) => {
    // =========== 关键修复：即使请求失败也要检查Token ===========
    if (response && response.header) {
      const newToken = response.header?.['new-token'] || response.header?.['New-Token'];
      if (newToken) {
        console.log('⚠️ 请求失败但收到新Token，更新', newToken.substring(0, 50) + '...');
        utils_storage.setToken(newToken);
      }
    }
    // =========== 关键修复结束 ===========
    
    const status = response?.statusCode;
    let message = "";
    console.log("response", response);
    
    switch (status) {
      case 401:
        message = "token 失效，请重新登录";
        // 清除无效的token
        utils_storage.removeToken();
        break;
      case 403:
        message = "拒绝访问";
        // 403也可能是token失效
        utils_storage.removeToken();
        break;
      case 404:
        message = "请求地址错误";
        break;
      case 500:
        message = "服务器故障";
        break;
      default:
        message = "网络连接故障";
    }
    
    common_vendor.index.showToast({
      title: message,
      icon: "error"
    });
    return Promise.reject(response);
  }
);
